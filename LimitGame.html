// script.js - Initial setup and game logic console.log("Edge Slider script loaded."); // --- DOM Element References --- const canvas = document.getElementById('gameCanvas'); const ctx = canvas.getContext('2d'); const inputSlider = document.getElementById('inputSlider'); const sliderValueDisplay = document.getElementById('sliderValue'); const launchButton = document.getElementById('launchButton'); const calculateButton = document.getElementById('calculateButton'); const calcInputX = document.getElementById('calcInputX'); const calcResultP = document.getElementById('calcResultP'); const roundNumberDisplay = document.getElementById('roundNumber'); const frictionValueDisplay = document.getElementById('frictionValue'); const limitValueDisplay = document.getElementById('limitValue'); const generalFormulaDisplay = document.getElementById('generalFormula'); const specificFormulaDisplay = document.getElementById('specificFormula'); const feedbackTextDisplay = document.getElementById('feedbackText'); const totalErrorScoreDisplay = document.getElementById('totalErrorScore'); // --- Event Listeners --- // Update slider value display dynamically inputSlider.addEventListener('input', () => { sliderValueDisplay.textContent = parseFloat(inputSlider.value).toFixed(2); }); // Named event handlers for adding/removing later function launchClickHandler() { const currentForceX = parseFloat(inputSlider.value); console.log("Launch button clicked. Input x:", currentForceX); if (!gameActive) { feedbackTextDisplay.textContent = "Launching..."; launchButton.disabled = true; // Disable button during launch calculateButton.disabled = true; // Disable calc button too animateLaunch(currentForceX, currentFriction, currentLimit); } else { console.log("Game is already active."); } } function nextRoundClickHandler() { console.log("Next Round button clicked."); setupNewRound(); // Setup the next round } // Calculate button action calculateButton.addEventListener('click', () => { const testX = parseFloat(calcInputX.value); console.log("Calculate button clicked. Test x:", testX); if (!isNaN(testX)) { // Use the actual calculation function const calculatedP = calculateStoppingPosition(testX, currentFriction, currentLimit); calcResultP.textContent = calculatedP.toFixed(2); feedbackTextDisplay.textContent = `Calculated P(${testX.toFixed(2)}) = ${calculatedP.toFixed(2)}. Ready to launch?`; } else { calcResultP.textContent = "--"; feedbackTextDisplay.textContent = "Please enter a valid number for x."; } }); // --- Game State Variables --- let currentRound = 1; let totalError = 0; let currentFriction = 0.1; let currentLimit = 100.0; let gameActive = false; // Flag to indicate if an animation is in progress // Initialize formula strings let generalFormulaString = ""; let specificFormulaString = ""; // --- Physics & Formula Definition --- // IMPORTANT: Define the actual physics model here. // This example uses a simple model where force creates initial velocity, // and constant friction force causes deceleration. // v_initial = k_force * x // a = -friction * g (assuming friction force is constant: F_fric = μ * N = μ * m * g) // Stopping distance P = v_initial^2 / (2 * |a|) = (k_force * x)^2 / (2 * friction * g) const k_force = 5; // Factor converting input 'x' to initial velocity const gravity = 9.8; // Assume standard gravity for friction calculation // Function representing the physics model to calculate final position // This is used by the Calculation Pad function calculateStoppingPosition(x, friction, limit /* ignored */) { if (friction <= 0) return Infinity; // Avoid division by zero, infinite travel const initialVelocity = k_force * x; const acceleration = -friction * gravity; // Deceleration due to friction if (acceleration >= 0) return Infinity; // No stopping force const stoppingDistance = (initialVelocity * initialVelocity) / (2 * Math.abs(acceleration)); console.log(`Calc P: x=${x}, μ=${friction}, v0=${initialVelocity.toFixed(2)}, a=${acceleration.toFixed(2)}, P=${stoppingDistance.toFixed(2)}`); return stoppingDistance; } // Function to get position P at a given time t during the slide // Used by the animation loop function getPositionAtTime(t, startX, friction) { const initialVelocity = k_force * startX; const acceleration = -friction * gravity; // Constant deceleration // Standard kinematic equation: P(t) = v0*t + 0.5*a*t^2 let position = initialVelocity * t + 0.5 * acceleration * t * t; // Calculate time to stop: v(t) = v0 + a*t => 0 = v0 + a*t_stop => t_stop = -v0/a const timeToStop = (acceleration !== 0) ? -initialVelocity / acceleration : Infinity; if (t >= timeToStop) { // If time is past stopping time, return the final stopping position position = calculateStoppingPosition(startX, friction, 0); // Limit not needed here } return position < 0 ? 0 : position; // Don't allow negative position } // Function to get velocity V at a given time t function getVelocityAtTime(t, startX, friction) { const initialVelocity = k_force * startX; const acceleration = -friction * gravity; let velocity = initialVelocity + acceleration * t; // Calculate time to stop const timeToStop = (acceleration !== 0) ? -initialVelocity / acceleration : Infinity; if (t >= timeToStop) { velocity = 0; // Velocity is zero after stopping } return velocity; } // --- Core Game Logic Functions --- // Sets up constants and UI for a new round function setupNewRound() { console.log(`Setting up Round ${currentRound}`); gameActive = false; // Ensure game is not marked active // Example: Slightly increase friction each round, reset limit currentFriction = 0.08 + currentRound * 0.02; currentLimit = 100.0 + (currentRound - 1) * 10; // Make it slightly harder // Update formula strings based on current constants // General formula explanation (conceptual) generalFormulaString = "P = v₀²/ (2 * μ * g) where v₀ = k * x"; // Specific formula string for display (using the calculated stopping distance formula) specificFormulaString = `P(x) = (${k_force}×x)² / (2 × ${currentFriction.toFixed(2)} × ${gravity})`; updateUI(); // Update displays with new round info drawBackground(currentLimit); // Redraw background with potentially new limit drawPuck(0); // Reset puck position visually } // Main animation loop using requestAnimationFrame function animateLaunch(startX, friction, limit) { if (gameActive) return; gameActive = true; console.log(`Animating launch with x=${startX}, μ=${friction}, L=${limit}`); let startTime = null; let currentPosition = 0; let currentTime = 0; let pathData = [{ t: 0, p: 0 }]; // Start trail at origin function animationStep(timestamp) { if (!startTime) startTime = timestamp; let elapsedTime = (timestamp - startTime) / 1000; // Time in seconds currentTime = elapsedTime; // --- Physics Calculation Step --- currentPosition = getPositionAtTime(currentTime, startX, friction); let currentVelocity = getVelocityAtTime(currentTime, startX, friction); // Add point to path data for trail pathData.push({ t: currentTime, p: currentPosition }); // --- Drawing Step --- ctx.clearRect(0, 0, canvas.width, canvas.height); // Clear canvas drawBackground(limit); // Draw background (axes, limit line) drawTrail(pathData); // Draw the trail drawPuck(currentPosition); // Draw the puck // --- Check Stop Conditions --- // Stop when velocity is zero (or effectively zero) let shouldStop = Math.abs(currentVelocity) <= 0.01; // Safety break if position goes way beyond limit (e.g., infinite travel if mu=0) if (currentPosition > limit * 1.5 || currentTime > 20) { // Added time limit console.warn("Stopping animation due to exceeding bounds or time limit."); shouldStop = true; // If stopped artificially, use the theoretical final position for scoring currentPosition = calculateStoppingPosition(startX, friction, limit); } if (!shouldStop && gameActive) { // Check gameActive flag in case of external stop requestAnimationFrame(animationStep); // Continue animation } else { // --- Animation End --- if (!gameActive && !shouldStop) { // If stopped externally (e.g. reset button) console.log("Animation stopped externally."); return; } gameActive = false; // Ensure the final point is added accurately const finalTime = (Math.abs(currentVelocity) <= 0.01 && friction > 0) ? -(k_force * startX) / (-friction * gravity) : currentTime; const finalPosition = calculateStoppingPosition(startX, friction, limit); pathData.push({ t: finalTime, p: finalPosition }); // Redraw one last time with the final point ctx.clearRect(0, 0, canvas.width, canvas.height); drawBackground(limit); drawTrail(pathData); drawPuck(finalPosition); console.log(`Animation stopped at t=${finalTime.toFixed(2)}s, P=${finalPosition.toFixed(2)}`); handleRoundEnd(finalPosition, limit); } } requestAnimationFrame(animationStep); // Start the animation loop } // Handle scoring and feedback after animation stops function handleRoundEnd(finalPosition, limit) { let distance = limit - finalPosition; let roundError = 0; let feedback = ""; // Handle potential Infinity case (e.g., friction is zero) if (!isFinite(finalPosition)) { roundError = 500; // Large penalty for infinite travel feedback = `Travelled infinitely (μ=${currentFriction})! Limit was ${limit.toFixed(1)}. Error penalty: ${roundError.toFixed(2)}`; feedbackTextDisplay.classList.add('text-red-600'); feedbackTextDisplay.classList.remove('text-green-600'); } else if (finalPosition > limit) { // Overshot roundError = 50 + (finalPosition - limit); // Penalty + overshoot amount feedback = `Overshot! Final Position: ${finalPosition.toFixed(2)}. Limit was ${limit.toFixed(1)}. Error penalty: ${roundError.toFixed(2)}`; feedbackTextDisplay.classList.add('text-red-600'); feedbackTextDisplay.classList.remove('text-green-600'); } else { // Success (at or before limit) roundError = distance; // Error is the distance remaining feedback = `Success! Final Position: ${finalPosition.toFixed(2)}. Distance from limit: ${distance.toFixed(2)}. Error: ${roundError.toFixed(2)}`; feedbackTextDisplay.classList.remove('text-red-600'); feedbackTextDisplay.classList.add('text-green-600'); // Award bonus points for closeness if (distance < 0.1) feedback += " Perfect!"; else if (distance < 1.0) feedback += " Excellent!"; else if (distance < 5.0) feedback += " Good!"; } totalError += roundError; feedbackTextDisplay.textContent = feedback; totalErrorScoreDisplay.textContent = totalError.toFixed(2); // Prepare for next round currentRound++; console.log("Round ended. Total Error:", totalError); // Change Launch button to "Next Round" and update listener launchButton.textContent = "Next Round"; launchButton.removeEventListener('click', launchClickHandler); // Remove old listener launchButton.addEventListener('click', nextRoundClickHandler); // Add new listener launchButton.disabled = false; // Re-enable button calculateButton.disabled = false; // Re-enable calc button } // --- Drawing Functions --- function drawBackground(limit) { ctx.fillStyle = '#f0f0f0'; // Match canvas background ctx.fillRect(0, 0, canvas.width, canvas.height); // Graph drawing parameters const padding = 40; const graphWidth = canvas.width - padding * 2; const graphHeight = canvas.height - padding * 2; const originX = padding; const originY = canvas.height - padding; // Determine graph scale based on limit const maxPosition = limit * 1.2; // Y-axis scale (Position) const maxTime = 5.0; // Fixed X-axis scale (Time in seconds), adjust as needed ctx.strokeStyle = '#9ca3af'; // Gray color for axes ctx.lineWidth = 1; ctx.fillStyle = '#374151'; // Dark gray for text ctx.font = '10px Inter'; // Y-Axis (Position) ctx.beginPath(); ctx.moveTo(originX, padding); ctx.lineTo(originX, originY); ctx.stroke(); ctx.fillText("Position (P)", originX - 35, padding + 5); // Y-axis labels/ticks for (let p = 0; p <= maxPosition; p += maxPosition / 5) { // 5 ticks example const tickY = originY - (p / maxPosition) * graphHeight; ctx.beginPath(); ctx.moveTo(originX - 5, tickY); ctx.lineTo(originX, tickY); ctx.stroke(); ctx.fillText(p.toFixed(0), originX - 30, tickY + 3); } // X-Axis (Time) ctx.beginPath(); ctx.moveTo(originX, originY); ctx.lineTo(canvas.width - padding, originY); ctx.stroke(); ctx.fillText("Time (t)", canvas.width - padding - 15, originY + 15); // X-axis labels/ticks for (let t = 0; t <= maxTime; t += maxTime / 5) { // 5 ticks example const tickX = originX + (t / maxTime) * graphWidth; ctx.beginPath(); ctx.moveTo(tickX, originY); ctx.lineTo(tickX, originY + 5); ctx.stroke(); ctx.fillText(t.toFixed(1), tickX - 5, originY + 15); } // Limit Line L const limitY = originY - (limit / maxPosition) * graphHeight; if (limitY >= padding && limitY <= originY) { // Only draw if within graph bounds (adjusted check) ctx.strokeStyle = '#ef4444'; // Red color for limit ctx.lineWidth = 1.5; ctx.setLineDash([4, 4]); // Dashed line ctx.beginPath(); ctx.moveTo(originX, limitY); ctx.lineTo(canvas.width - padding, limitY); ctx.stroke(); ctx.setLineDash([]); // Reset line dash ctx.fillStyle = '#ef4444'; ctx.fillText(`Limit L=${limit.toFixed(1)}`, originX + 5, limitY - 5); } } function drawPuck(position) { // Scales position P to canvas Y-coordinate const padding = 40; const graphHeight = canvas.height - padding * 2; const originY = canvas.height - padding; const maxPosition = currentLimit * 1.2; // Use currentLimit for scaling let puckY = originY - (position / maxPosition) * graphHeight; // Clamp puck position within visible graph area puckY = Math.max(padding - 8, Math.min(originY + 8, puckY)); // Allow slight overlap // Keep puck visually aligned with the Y-axis const puckX = padding; ctx.fillStyle = '#3b82f6'; // Blue color for puck ctx.beginPath(); ctx.arc(puckX, puckY, 6, 0, Math.PI * 2); // Draw a circle for the puck ctx.fill(); ctx.strokeStyle = '#2563eb'; // Darker blue border ctx.lineWidth = 1; ctx.stroke(); } function drawTrail(pathData) { if (pathData.length < 2) return; const padding = 40; const graphWidth = canvas.width - padding * 2; const graphHeight = canvas.height - padding * 2; const originX = padding; const originY = canvas.height - padding; const maxPosition = currentLimit * 1.2; // Use the same fixed max time as the background axis for consistent scaling const maxTime = 5.0; ctx.strokeStyle = '#10b981'; // Green color for trail ctx.lineWidth = 2; ctx.beginPath(); // Use a flag to track if we need to use moveTo instead of lineTo let needsMove = true; pathData.forEach((point, index) => { // Scale time and position to canvas coordinates const xCoord = originX + (point.t / maxTime) * graphWidth; let yCoord = originY - (point.p / maxPosition) * graphHeight; // Clamp coordinates to prevent drawing outside graph area (padding) const clampedX = Math.max(originX, Math.min(canvas.width - padding, xCoord)); const clampedY = Math.max(padding, Math.min(originY, yCoord)); if (index === 0 || needsMove) { ctx.moveTo(clampedX, clampedY); needsMove = false; } else { // Check if the point is reasonably within canvas bounds before drawing line segment // This helps prevent huge lines if a calculation error results in extreme values if (isFinite(clampedX) && isFinite(clampedY) && Math.abs(clampedX - originX) < graphWidth * 1.5 && Math.abs(clampedY - originY) < graphHeight * 1.5) { ctx.lineTo(clampedX, clampedY); } else { // If point is way off, just move to it to avoid drawing a long stray line ctx.moveTo(clampedX, clampedY); needsMove = true; } } }); ctx.stroke(); } // Updates all UI text elements with current game state function updateUI() { roundNumberDisplay.textContent = currentRound; frictionValueDisplay.textContent = currentFriction.toFixed(2); limitValueDisplay.textContent = currentLimit.toFixed(1); generalFormulaDisplay.textContent = generalFormulaString; specificFormulaDisplay.textContent = specificFormulaString; totalErrorScoreDisplay.textContent = totalError.toFixed(2); sliderValueDisplay.textContent = parseFloat(inputSlider.value).toFixed(2); calcResultP.textContent = "--"; calcInputX.value = ""; // Reset button states and feedback text/color for new round launchButton.disabled = false; calculateButton.disabled = false; launchButton.textContent = "Launch"; feedbackTextDisplay.classList.remove('text-red-600', 'text-green-600'); feedbackTextDisplay.textContent = `Round ${currentRound}: Adjust input and launch.`; // Ensure correct event listener is on launch button launchButton.removeEventListener('click', nextRoundClickHandler); launchButton.removeEventListener('click', launchClickHandler); // Remove first to avoid duplicates launchButton.addEventListener('click', launchClickHandler); } // --- Initial Game Setup --- function initializeGame() { console.log("Initializing game..."); // Remove any existing event listeners to avoid duplicates launchButton.removeEventListener('click', launchClickHandler); launchButton.removeEventListener('click', nextRoundClickHandler); // Add the primary launch listener initially launchButton.addEventListener('click', launchClickHandler); // Setup the first round (which also calls updateUI) setupNewRound(); // Initial draw of the background and puck drawBackground(currentLimit); drawPuck(0); console.log("Game Initialized and Ready."); } // Start the game initialization process once the script loads initializeGame();
