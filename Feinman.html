import matplotlib.pyplot as plt import matplotlib.patches as patches # Set up the figure and axis fig, ax = plt.subplots(figsize=(6, 4)) # Define the vertex position (center of the diagram) vertex = (0, 0) # Define the endpoints of the three lines (incoming and outgoing particles) # Incoming particle 1 (mode n): from bottom-left to vertex line1_start = (-1, -1) line1_end = vertex # Incoming particle 2 (mode m): from bottom-right to vertex line2_start = (1, -1) line2_end = vertex # Outgoing particle (mode -n-m): from vertex to top line3_start = vertex line3_end = (0, 1) # Draw the lines with arrows # Incoming line 1 (mode n) ax.arrow(line1_start[0], line1_start[1], line1_end[0] - line1_start[0], line1_end[1] - line1_start[1], head_width=0.05, head_length=0.1, fc='black', ec='black', length_includes_head=True) # Incoming line 2 (mode m) ax.arrow(line2_start[0], line2_start[1], line2_end[0] - line2_start[0], line2_end[1] - line2_start[1], head_width=0.05, head_length=0.1, fc='black', ec='black', length_includes_head=True) # Outgoing line (mode -n-m) ax.arrow(line3_start[0], line3_start[1], line3_end[0] - line3_start[0], line3_end[1] - line3_start[1], head_width=0.05, head_length=0.1, fc='black', ec='black', length_includes_head=True) # Highlight the vertex with a red dot ax.plot(vertex[0], vertex[1], 'ro', markersize=10) # Add labels for the mode numbers # Label for mode n (near the incoming line 1) ax.text(-0.7, -0.5, r'$n$', fontsize=12, color='blue') # Label for mode m (near the incoming line 2) ax.text(0.7, -0.5, r'$m$', fontsize=12, color='blue') # Label for mode -n-m (near the outgoing line) ax.text(0.2, 0.5, r'$-n-m$', fontsize=12, color='blue') # Set axis limits and remove axes for a clean diagram ax.set_xlim(-1.5, 1.5) ax.set_ylim(-1.5, 1.5) ax.axis('off') # Add a title plt.title('Feynman Diagram: Kaluza-Klein Interaction', fontsize=14) # Show the plot plt.show()
